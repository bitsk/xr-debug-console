cmake_minimum_required(VERSION 3.16)
project(xr_debug_helpers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(XRDevice CONFIG REQUIRED)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(VULKAN REQUIRED IMPORTED_TARGET vulkan)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-allocators-1.0)
find_library(YUV_LIBRARY NAMES yuv REQUIRED)
find_library(TURBOJPEG_LIBRARY NAMES turbojpeg REQUIRED)
find_path(OPENVR_INCLUDE_DIR openvr.h HINTS ${XR_OPENVR_INCLUDE_DIR} PATH_SUFFIXES openvr REQUIRED)
find_program(GLSLANG_COMMAND NAMES glslangValidator glslang REQUIRED)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(PRIVATE "${CMAKE_INSTALL_LIBDIR}/xr/debug")

add_executable(nv12-capture ${ROOT}/video/nv12-capture.c)
target_link_libraries(nv12-capture PRIVATE PkgConfig::DRM ${YUV_LIBRARY})

add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shader.h
  COMMAND ${GLSLANG_COMMAND} -V --target-env vulkan1.1 ${ROOT}/video/gpu/rgba-to-nv12.comp -o ${CMAKE_CURRENT_BINARY_DIR}/shader.spv
  COMMAND ${Python3_EXECUTABLE} ${ROOT}/tools/embed-spirv.py ${CMAKE_CURRENT_BINARY_DIR}/shader.spv ${CMAKE_CURRENT_BINARY_DIR}/shader.h
  DEPENDS ${ROOT}/video/gpu/rgba-to-nv12.comp ${ROOT}/tools/embed-spirv.py)
add_executable(nv12-capture-gpu ${ROOT}/video/gpu/nv12-capture-gpu.c ${CMAKE_CURRENT_BINARY_DIR}/shader.h)
target_include_directories(nv12-capture-gpu PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(nv12-capture-gpu PRIVATE PkgConfig::DRM PkgConfig::VULKAN PkgConfig::GST)
install(TARGETS nv12-capture nv12-capture-gpu RUNTIME DESTINATION ${PRIVATE}/video)

add_executable(capture ${ROOT}/video/kms/capture.c)
target_link_libraries(capture PRIVATE PkgConfig::DRM ${TURBOJPEG_LIBRARY})
install(TARGETS capture RUNTIME DESTINATION ${PRIVATE})
foreach(name camera-imx vr-timing)
  add_executable(${name} ${ROOT}/helpers/${name}.cpp)
  target_include_directories(${name} PRIVATE ${OPENVR_INCLUDE_DIR})
  target_link_libraries(${name} PRIVATE XRDevice::api pthread dl)
  install(TARGETS ${name} RUNTIME DESTINATION ${PRIVATE}/tools)
endforeach()
